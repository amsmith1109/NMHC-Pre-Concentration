% Turbo Pump
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\pgfdeclareshape{turbo pump}{%
    \inheritsavedanchors[from=circle]%
    \inheritanchorborder[from=circle]%
    \inheritanchor[from=circle]{center}%
    \inheritanchor[from=circle]{radius}%
    \anchor{inlet}{
        \pgf@x=0pt
        \pgf@y=1.5\radius
    }
    \anchor{outlet}{
        \pgf@x=0pt
        \pgf@y=-1.5\radius    
    }  
  \backgroundpath{%
    \pgf@xa\radius% store radius in auxiliary length 
    % Outline
    \pgfpathcircle{\pgfpoint{0cm}{0cm}}{\pgf@xa}
    
    % Inlet and Outlet Lines
    \pgfpathmoveto{\pgfpoint{0pt}{\pgf@xa}}%
    \pgfpathlineto{\pgfpoint{0pt}{1.5\pgf@xa}}
    \pgfpathmoveto{\pgfpoint{0pt}{-1\pgf@xa}}%
    \pgfpathlineto{\pgfpoint{0pt}{-1.5\pgf@xa}}
    
    % Angled Lines
    \pgfpathmoveto{\pgfpoint{-0.4358\pgf@xa}{-0.9\pgf@xa}}%
    \pgfpathlineto{\pgfpoint{-0.8\pgf@xa}{0.6\pgf@xa}}%
    \pgfpathmoveto{\pgfpoint{0.4358\pgf@xa}{-0.9\pgf@xa}}%
    \pgfpathlineto{\pgfpoint{0.8\pgf@xa}{0.6\pgf@xa}}%

    % Turbo Specific Shape
    \pgfpathmoveto{\pgfpoint{0.3\pgf@xa}{0cm}}
    \pgfpatharc{0}{360}{0.3\pgf@xa}
    \pgfpathmoveto{\pgfpoint{0.15\pgf@xa}{0cm}}
    \pgfpatharc{0}{360}{0.15\pgf@xa}
    % \pgfpathcircle{\pgfpoint{0cm}{0cm}}{0.2\pgf@xa}
    }
}
